合约设计（SkyHedgeCore.sol）

状态机

状态流转
// BIDDING → ACTIVE → PAID（延误触发）
//                  → EXPIRED（未延误，抵押返还）

enum Status { BIDDING, ACTIVE, PAID, EXPIRED }

核心数据结构

struct Policy
struct Policy {
    bytes32 flightRef;         // 航班号
    uint256 departureTime;     // 出发时间（Unix timestamp）
    uint256 delayThreshold;    // 延误触发阈值（分钟）
    uint256 fixedPayout;       // 赔付金额（wei）
    uint256 auctionEnd;        // 拍卖截止时间
    uint256 expiry;            // 保单过期时间
    uint256 bestPremium;       // 当前最低保费
    address bestUnderwriter;   // 当前最低报价方
    Status  status;
    uint256 policyNFTId;       // 乘客持有的NFT tokenId
    uint256 riskNFTId;         // 承保人持有的NFT tokenId
}

四个核心函数签名

核心接口
// 1. 乘客创建保单
function createPolicy(
    bytes32 flightRef,
    uint256 departureTime,
    uint256 delayThreshold,
    uint256 fixedPayout,
    uint256 auctionEnd,
    uint256 expiry
) external returns (uint256 policyId);

// 2. 承保人报价（逆向拍卖）
function bidPremium(uint256 policyId, uint256 premiumAmount) external;

// 3. 拍卖结束，锁定抵押，mint两个NFT
function finalizeAuction(uint256 policyId) external payable nonReentrant;
// ↑ msg.value 必须等于 fixedPayout（全额抵押）

// 4. Oracle推送延误数据，自动结算
function resolvePolicy(uint256 policyId, uint256 delayMins) external nonReentrant;
// ↑ 只有 resolver 地址能调用

安全设计要点（评分项）

安全机制用于哪个函数作用
require(msg.sender == resolver)resolvePolicy防止任意人触发结算
require(msg.value == fixedPayout)finalizeAuction强制全额抵押
ReentrancyGuard所有涉及ETH转账的函数防重入攻击
Checks-Effects-InteractionsfinalizeAuction / resolvePolicy先改状态，再转账
require(block.timestamp > auctionEnd)finalizeAuction防止提前结束拍卖

前端设计（单页三Tab）

⚡
三个Tab是逻辑分区，实现上优先单页。不用Router，不用状态管理库，一个 App.jsx 搞定。

钱包连接（不引入第三方库）

wallet connection · 约20行
// 直接用 ethers.js，不需要 wagmi/rainbowkit/web3modal
const connectWallet = async () => {
  const provider = new ethers.BrowserProvider(window.ethereum);
  const signer = await provider.getSigner();
  const contract = new ethers.Contract(CONTRACT_ADDRESS, ABI, signer);
  setContract(contract);
  setSigner(signer);
};

三个Tab的核心功能

Tab角色核心操作
Passenger 乘客（账号A） 填写航班信息 → createPolicy() → 查看Policy NFT → （亮点）转让Policy NFT给账号C
Underwriter 承保人（账号B） 浏览待投标保单 → bidPremium() → finalizeAuction()（含锁定0.01ETH抵押）
Resolver Oracle（账号C/管理员） 输入policyId + 延误分钟 → resolvePolicy() → 显示结算结果

账号B · 继续
完成拍卖，锁定抵押

乘客转让 Policy NFT
